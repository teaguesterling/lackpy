import json
from pathlib import Path


def load_valid_rows(text):
    rows = []
    for line in text.splitlines():
        try:
            row = json.loads(line)
        except json.JSONDecodeError:
            continue
        rows.append(row)
    return rows


def main():
    path = Path("data/rows.jsonl")
    text = path.read_text()
    rows = load_valid_rows(text)
    print(len(rows))


main()
