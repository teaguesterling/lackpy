```python
# /// intent
# goal   = "Classify each valid row's size as small, medium, or large and print it in file order."
# reads  = ["data/rows.jsonl"]
# writes = []
# output = "printed"
# ///

import json
from pathlib import Path


def classify_size(size):
    if size < 100:
        return "small"
    if size < 200:
        return "medium"
    return "large"


text = Path("data/rows.jsonl").read_text()
for line in text.splitlines():
    try:
        row = json.loads(line)
    except json.JSONDecodeError:
        continue
    size = row["size"]
    print(classify_size(size))
```
