```python
# /// intent
# goal   = "Update the version in data/config.json to 1.3.0 and write it back with indent 2"
# reads  = ["data/config.json"]
# writes = ["data/config.json"]
# output = "none"
# ///

import json
from pathlib import Path

def load_config():
    text = Path('data/config.json').read_text()
    return json.loads(text)

def write_config(obj):
    Path('data/config.json').write_text(json.dumps(obj, indent=2))

def main():
    config = load_config()
    config['version'] = '1.3.0'
    write_config(config)

main()
```
