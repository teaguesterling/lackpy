```python
# /// intent
# goal   = "Count the rows in data/rows.jsonl by status, skipping invalid JSON lines, and print each status and its count sorted by status"
# reads  = ["data/rows.jsonl"]
# writes = []
# output = "printed"
# ///

import json
from pathlib import Path
from collections import Counter

# Read the rows.jsonl file
path = Path("data/rows.jsonl")
lines = path.read_text().splitlines()

# Count status by parsing valid JSON lines
status_counter = Counter()

for line in lines:
    line = line.strip()
    if not line:
        continue
    try:
        data = json.loads(line)
        status = data.get("status")
        if status is not None:
            status_counter[status] += 1
    except json.JSONDecodeError:
        continue

# Sort statuses and print each status and its count
for status in sorted(status_counter):
    print(f"{status}: {status_counter[status]}")
```
